mutation_data <- read.csv("COSMIC_mutations.csv", header = TRUE, stringsAsFactors = FALSE)
# Summarize VUDR as a single category
vudr_total <- mutation_data %>%
  filter(Mutation_type == "VUDR") %>%
  summarise(Combined_count = sum(Combined_count)) %>%
  mutate(Collapsed = "VUDR",
         Mutation_type = "VUDR")

# Calculate the percentage of VUDR
vudr_percentage <- round(vudr_total$Combined_count / sum(mutation_data$Combined_count) * 100, 1)

# Remove individual VUDR mutations and add the summarized VUDR category
plot_data <- mutation_data %>%
  filter(Mutation_type != "VUDR") %>%
  bind_rows(vudr_total) %>%
  arrange(Combined_count) %>%  # Sort by increasing size
  mutate(
    pie_label = ifelse(Collapsed == "VUDR", paste0("VUDR\n", vudr_percentage, "%"), Collapsed),  # Label for pie
    legend_label = factor(Collapsed, levels = c("VUDR", unique(Collapsed[Collapsed != "VUDR"])))  # Fix factor levels
  )

# Generate colors using the "Spectral" palette
getPalette <- colorRampPalette(rev(brewer.pal(9, "Spectral")))
palette_colors <- getPalette(length(unique(plot_data$Collapsed)))

# Assign colors: VUDR is "lightgrey", others use the generated palette
mutation_colors <- setNames(palette_colors, unique(plot_data$Collapsed[plot_data$Collapsed != "VUDR"]))
mutation_colors["VUDR"] <- "lightgrey"  # Explicitly set VUDR to light grey
plot <- ggplot(plot_data, aes(x = "", y = Combined_count, fill = legend_label)) +
  geom_bar(stat = "identity", width = 0.5, color = "black", size = 0.03) +  # Add grey edges
  coord_polar("y", start = 0) +
  scale_fill_manual(values = mutation_colors) +  # Apply custom colors
  theme_void() +
  geom_text(aes(label = ifelse(Collapsed == "VUDR", paste0(vudr_percentage, "%"), "")), 
            position = position_stack(vjust = 0.5), size = 5) +  # Display VUDR % inside the pie
  theme(legend.title = element_blank(),
        legend.position = "right") #+
